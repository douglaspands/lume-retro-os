# USB device role on the OTG port (usb-device-access D3, D4): MTP server for ROMS and the
# SSH server of the USB network, both started by lume-usb (the dropbear service that
# Alpine adds with OpenRC is not enabled)
umtprd
dropbear
# udhcpd of the USB network: in Alpine it is in busybox-extras, not in busybox
busybox-extras

# Wi-Fi and Bluetooth by USB dongle (usb-device-access D9, D10): NetworkManager drives the
# Wi-Fi menu of RetroArch (nmcli; it starts wpa_supplicant through D-Bus), bluez the
# Bluetooth menu (bluetoothctl). The dongle firmware is hardware: board/<console>/hw-packages.txt.
dbus
dbus-openrc
networkmanager
networkmanager-openrc
networkmanager-wifi
networkmanager-cli
wpa_supplicant
bluez
bluez-openrc

# Core scripts and settings (aports/lume-core, a noarch fragment): logs, session, frontend
# service, hostname/issue/motd
lume-core

# The rest of the former overlay (noarch fragments, build-speed D2): USB, network, audio, saves,
# diagnostics, power, game catalog, launcher data, RetroArch settings, shaders
lume-usb
lume-net
lume-audio
lume-saves
lume-diag
lume-power
lume-games
lume-launcher-data
lume-retroarch-cfg
# lume-shaders: out of the image until they run at full speed on the GA36 (P-89; 26.7 fps with
# gba-modern-remaster on lima, 2026-11-17)

# Frontend
retroarch
libretro-core-info
retroarch-assets
retroarch-joypad-autoconfig

# Input daemon (lume-launcher D8, aports/lume-input): one virtual gamepad from the built-in
# controls, Fn+Start for the launcher and the volume buttons, with a profile per board
lume-input

# Load controller of the power profiles (game-governor D12, aports/lume-powerctl): started by
# lume-power (lume-power) in the game profiles with controller=load
lume-powerctl

# The sampling loop of the diagnostic logs in C (game-governor 1.2, aports/lume-logd): lume-logd
# (lume-core) execs it after the hardware snapshot; without it the shell loop runs
lume-logd

# The default frontend (lume-launcher, aports/lume-launcher): interface, game lists and
# supervisor
lume-launcher

# Runner "love" of the launcher (lume-launcher D5): LÖVE 11.5 from Alpine v3.24, packaged
# for x86_64, armv7 and aarch64 (task 1.3), so no local aport is needed
love
